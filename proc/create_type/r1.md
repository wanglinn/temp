以下是基于 PL/SQL 匿名块（`EXEC SQL EXECUTE ... END-EXEC`）的 scanner 收集模式，为 `CREATE TYPE` 和 `CREATE TYPE BODY` 设计的完整透传方案。核心思路是：**Parser 识别 `CREATE TYPE` / `CREATE TYPE BODY` 后，触发 Scanner 进入 `xctype` 文本收集模式，按不同结束条件退出，文本原样透传 XDB。**

---

## 一、实现概要

| 语句 | 结束条件 | 嵌套计数 | 说明 |
|---|---|---|---|
| `CREATE TYPE` 声明 | `;`（分号） | 无 | 声明内部无 `BEGIN/END`，直接到 `;` 结束 |
| `CREATE TYPE BODY` | 最后的 `END;` | 需 `BEGIN/END` 深度计数 | 成员函数含 `BEGIN...END;`，需区分 `END IF`/`END LOOP` 等控制流 |

---

3. **`ecpg.tokens`** — 新增 `CTYPE_TEXT` token
4. **`preproc_extern.h`** — 声明外部标志变量

## 三、Scanner 层 (`pgc.l`)

### 新增变量与模式声明

在 `pgc.l` 的全局区（`%{ ... %}` 内）新增：

```c
/* 触发 CREATE TYPE 文本收集模式 */
bool enter_ctype_mode = false;
bool ctype_is_body = false;
int  ctype_begin_depth = 0;
```

在 `%x` 声明区新增 `xctype`：

```c
%x xplsql
%x xctype    /* 新增：CREATE TYPE / CREATE TYPE BODY 文本收集 */
```

### `yylex` 入口检查

在 `yylex` 的 `/* code to execute during start of each call */` 段中，加入：

```c
/* 如果 parser 触发了 CREATE TYPE 收集模式，切换进入 */
if (enter_ctype_mode)
{
    enter_ctype_mode = false;
    ctype_begin_depth = 0;
    BEGIN(xctype);
    startlit();
}
```

### 新增 `BODY` 关键字（SQL 模式）

如果 `pgc.l` 中 `<SQL>` 模式下没有 `BODY`，补充：

```c
<SQL>[Bb][Oo][Dd][Yy]		{ return BODY; }
```

### `xctype` 模式规则

```c
<xctype>{

/* 1. 字符串字面量：跳过内部的 ; 和 END */
\'				{
					addlit(yytext, yyleng);
					int c;
					while ((c = input()) != EOF)
					{
						if (c == '\\')
						{
							int c2 = input();
							if (c2 != EOF)
							{
								char buf[2] = {c, c2};
								addlit(buf, 2);
							}
						}
						else if (c == '\'')
						{
							addlit(&c, 1);
							break;
						}
						else
						{
							addlit(&c, 1);
						}
					}
				}

/* 2. 行注释 */
"--"			{
					addlit(yytext, yyleng);
					int c;
					while ((c = input()) != EOF && c != '\n')
						addlit(&c, 1);
					if (c == '\n')
						addlit(&c, 1);
				}

/* 3. 块注释 */
"/*"			{
					addlit(yytext, yyleng);
					int c, prev = 0;
					while ((c = input()) != EOF)
					{
						addlit(&c, 1);
						if (prev == '*' && c == '/')
							break;
						prev = c;
					}
				}

/* 4. 控制流 END：不触发深度计数，直接收集 */
[Ee][Nn][Dd][ \t\n]+[Ii][Ff]		{ addlit(yytext, yyleng); }
[Ee][Nn][Dd][ \t\n]+[Ll][Oo][Oo][Pp]	{ addlit(yytext, yyleng); }
[Ee][Nn][Dd][ \t\n]+[Cc][Aa][Ss][Ee]	{ addlit(yytext, yyleng); }
[Ee][Nn][Dd][ \t\n]+[Ff][Oo][Rr]	{ addlit(yytext, yyleng); }

/* 5. 单独的 BEGIN：增加嵌套深度（仅 TYPE BODY） */
[Bb][Ee][Gg][Ii][Nn]	{
					if (ctype_is_body)
						ctype_begin_depth++;
					addlit(yytext, yyleng);
				}

/* 6. 单独的 END：深度管理或退出 */
[Ee][Nn][Dd]	{
					if (ctype_is_body)
					{
						if (ctype_begin_depth > 0)
						{
							ctype_begin_depth--;
							addlit(yytext, yyleng);
						}
						else
						{
							/* TYPE BODY 结束，把 END 放回，由 <SQL> 模式返回 */
							yyless(0);
							BEGIN(SQL);
							base_yylval.str = mm_strdup(literalbuf);
							startlit();
							return CTYPE_TEXT;
						}
					}
					else
					{
						/* CREATE TYPE 声明中，END 只是普通文本 */
						addlit(yytext, yyleng);
					}
				}

/* 7. 分号：非 BODY 模式下语句结束 */
;				{
					if (!ctype_is_body)
					{
						yyless(0);	/* 放回 ;，由 <SQL> 匹配 */
						BEGIN(SQL);
						base_yylval.str = mm_strdup(literalbuf);
						startlit();
						return CTYPE_TEXT;
					}
					addlit(yytext, yyleng);
				}

/* 8. 其他所有字符：原样收集 */
.|\n			{
					addlit(yytext, yyleng);
				}

/* 9. 未终止错误 */
<<EOF>>			{
					mmfatal(PARSE_ERROR, "unterminated CREATE TYPE statement");
				}

} /* <xctype> */
```

## 四、Parser 层 (`preproc.y` / `ecpg.trailer`)

### 新增 Token 声明

在 `ecpg.tokens` 或 `preproc.y` 的 `%token` 区：

```yacc
%token <str> CTYPE_TEXT
```

### 新增语法规则

在 `statement` 规则中新增 `CreateTypeStmt` 分支（与 `ECPGPlsqlBlock` 并列）：

```yacc
| CreateTypeStmt
```

新增规则定义：

```yacc
/* CREATE TYPE / CREATE TYPE BODY 透传 */
CreateTypeStmt:
	ecpgstart CREATE OptOrReplace TYPE
	{
		enter_ctype_mode = true;
		ctype_is_body = false;
		ctype_begin_depth = 0;
	}
	CTYPE_TEXT ';'
	{
		char *stmt = cat_str(5,
			mm_strdup("CREATE "),
			$3,					/* OptOrReplace: "" 或 "OR REPLACE " */
			mm_strdup("TYPE "),
			$6,					/* CTYPE_TEXT 收集的文本 */
			mm_strdup(";"));
		output_statement(stmt, 1, ECPGst_normal);
		$$ = mm_strdup("");
	}
|	ecpgstart CREATE OptOrReplace TYPE BODY
	{
		enter_ctype_mode = true;
		ctype_is_body = true;
		ctype_begin_depth = 0;
	}
	CTYPE_TEXT END ';'
	{
		char *stmt = cat_str(7,
			mm_strdup("CREATE "),
			$3,					/* OptOrReplace */
			mm_strdup("TYPE BODY "),
			$7,					/* CTYPE_TEXT 收集的文本（不含最后的 END） */
			mm_strdup(" END"),
			mm_strdup(";"));
		output_statement(stmt, 1, ECPGst_normal);
		$$ = mm_strdup("");
	}
;

/* OR REPLACE 可选 */
OptOrReplace:
	/* empty */
	{ $$ = mm_strdup(""); }
|	OR REPLACE
	{ $$ = mm_strdup("OR REPLACE "); }
;
```

### 关于原生 `CREATE TYPE` 规则

**必须移除或注释掉** `preproc.y` 中原生 PostgreSQL 的 `CREATE TYPE` 语法规则（如 `CREATE TYPE name AS (field int)`、`CREATE TYPE name AS ENUM` 等），否则 Bison 会产生移进-归约冲突。

## 六、生成代码示例

### 输入 `.pc` 代码

```c
EXEC SQL CREATE OR REPLACE TYPE num_array AS TABLE OF NUMBER;

EXEC SQL CREATE TYPE BODY emp_t AS
  MEMBER FUNCTION get_name RETURN VARCHAR2 IS
  BEGIN
    RETURN name;
  END;
END;
```

### 预编译后生成的 C 代码（概念）

```c
/* 第一条 */
{ ECPGdo(__LINE__, 0, 1, NULL, 0, ECPGst_normal, 
  "CREATE OR REPLACE TYPE num_array AS TABLE OF NUMBER ;", 
  ECPGt_EOIT, ECPGt_EORT); }

/* 第二条 */
{ ECPGdo(__LINE__, 0, 1, NULL, 0, ECPGst_normal, 
  "CREATE TYPE BODY emp_t AS\n  MEMBER FUNCTION get_name RETURN VARCHAR2 IS\n  BEGIN\n    RETURN name;\n  END;\nEND ;", 
  ECPGt_EOIT, ECPGt_EORT); }
```

> 注意：第二条中 `CTYPE_TEXT` 收集到 `END` 之前的所有文本，然后 parser 拼接时补上 `END;`。

## 七、关键设计说明

| 问题 | 解决方式 |
|---|---|
| **字符串中的 `;` 误截断** | `xctype` 模式优先匹配 `\'` 字符串规则，内部字符全部收集，不触发 `;` 退出 |
| **注释中的 `;` 或 `END` 误触发** | 行注释 `--` 和块注释 `/* */` 规则优先级高于 `;` 和 `END` |
| **`END IF` 误减深度** | 复合规则 `[Ee][Nn][Dd][ \t\n]+[Ii][Ff]` 优先级高于单独 `END`，直接收集不处理 |
| **TYPE BODY 成员函数嵌套 `BEGIN/END`** | `ctype_begin_depth` 计数，只有 `begin_depth==0` 时的 `END` 才是 TYPE BODY 结束 |
| **原生 ECPG `CREATE TYPE` 冲突** | 移除原生 `preproc.y` 中 PG 语法的 `CREATE TYPE` 规则，全部由 `CreateTypeStmt` 接管 |

## 一句话总结

> **参照 PL/SQL 块的 `xplsql` 模式，新增 `xctype` 收集模式。`CREATE TYPE` 声明以 `;` 结束直接透传；`CREATE TYPE BODY` 维护 `BEGIN/END` 嵌套深度，忽略 `END IF` 等控制流，当 `begin_depth==0` 时遇到 `END` 即退出。Parser 层替换原生 `CREATE TYPE` 规则，文本拼接后走 `output_statement` 直接发给 XDB。**
